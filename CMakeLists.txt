cmake_minimum_required(VERSION 3.20)
project(FSTR_LUT VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(FSTR_SANITIZE "Enable address/undefined sanitizers" OFF)
add_library(fstr_core src/core/pipeline.cpp src/restoration/ffdnet.cpp src/looks/lut.cpp src/color/transfer.cpp src/color/grade.cpp src/color/profiles.cpp src/color/frame.cpp src/color/pixel.cpp)
target_include_directories(fstr_core PUBLIC include PRIVATE third_party)
target_link_libraries(fstr_core PUBLIC Threads::Threads)
add_executable(fstr_color_fixture tools/benchmark/color_fixture.cpp)
target_link_libraries(fstr_color_fixture PRIVATE fstr_core)
add_executable(fstr_color_render tools/benchmark/color_main.cpp)
target_link_libraries(fstr_color_render PRIVATE fstr_core)
if(MSVC)
  target_compile_options(fstr_core PRIVATE /W4)
else()
  target_compile_options(fstr_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
set(FSTR_AE_SDK_ROOT "" CACHE PATH "External After Effects SDK root (contains Examples)")
if(FSTR_AE_SDK_ROOT)
  if(NOT APPLE)
    message(FATAL_ERROR "Native bundle build currently verified only for macOS; Windows adapter packaging pending")
  endif()
  if(CMAKE_OSX_ARCHITECTURES AND NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "This native development bundle is arm64 only")
  endif()
  set(FSTR_AE_EXAMPLES "${FSTR_AE_SDK_ROOT}/Examples")
  if(NOT EXISTS "${FSTR_AE_EXAMPLES}/Headers/AE_Effect.h")
    message(FATAL_ERROR "External Adobe SDK headers missing")
  endif()
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/assets/looks/warm-film-v1.cube" FSTR_WARM_HASH)
  if(NOT FSTR_WARM_HASH STREQUAL "ee9e15323b80514ab031a97c9c7d1583132dd96d49bfb6a8e9a6cc43077521b5")
    message(FATAL_ERROR "Frozen Warm Film v1 bytes changed; do not alter existing-project looks")
  endif()
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/looks/warm-film-v1.cube" FSTR_WARM_CUBE)
  set(FSTR_SOURCE_BINDING "${CMAKE_CXX_COMPILER_ID};${CMAKE_CXX_COMPILER_VERSION};${CMAKE_BUILD_TYPE};${CMAKE_OSX_ARCHITECTURES};${CMAKE_CXX_FLAGS};${FSTR_SANITIZE}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/looks/warm-film-v1.cube")
  foreach(FSTR_COLOR_SOURCE src/ae/plugin.cpp src/ae/plugin.hpp src/ae/FSTR_LUT_PiPL.r src/ae/Info.plist.in src/ae/embedded_look.hpp.in include/fstr/color.hpp include/fstr/core.hpp src/color/pixel.cpp src/color/profiles.cpp src/color/grade.cpp src/color/transfer.cpp src/looks/lut.cpp CMakeLists.txt)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${FSTR_COLOR_SOURCE}" FSTR_SOURCE_HASH)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${FSTR_COLOR_SOURCE}")
    string(APPEND FSTR_SOURCE_BINDING "${FSTR_COLOR_SOURCE}:${FSTR_SOURCE_HASH};")
  endforeach()
  foreach(FSTR_SDK_HEADER AE_Effect.h AE_EffectVers.h AE_EffectCBSuites.h)
    file(SHA256 "${FSTR_AE_EXAMPLES}/Headers/${FSTR_SDK_HEADER}" FSTR_SDK_HASH)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${FSTR_AE_EXAMPLES}/Headers/${FSTR_SDK_HEADER}")
    string(APPEND FSTR_SOURCE_BINDING "SDK:${FSTR_SDK_HEADER}:${FSTR_SDK_HASH};")
  endforeach()
  string(SHA256 FSTR_COLOR_BUILD_ID "${FSTR_SOURCE_BINDING}")
  string(SUBSTRING "${FSTR_COLOR_BUILD_ID}" 0 16 FSTR_COLOR_BUILD_ID)
  configure_file(src/ae/embedded_look.hpp.in generated/fstr_embedded_look.hpp @ONLY)
  configure_file(src/ae/Info.plist.in generated/Info.plist @ONLY)
  set_target_properties(fstr_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_library(fstr_ae_objects OBJECT src/ae/plugin.cpp)
  target_link_libraries(fstr_ae_objects PRIVATE fstr_core)
  target_include_directories(fstr_ae_objects PRIVATE src/ae "${CMAKE_CURRENT_BINARY_DIR}/generated" "${FSTR_AE_EXAMPLES}/Headers" "${FSTR_AE_EXAMPLES}/Headers/SP" "${FSTR_AE_EXAMPLES}/Util")
  set_target_properties(fstr_ae_objects PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
  add_library(fstr_ae MODULE $<TARGET_OBJECTS:fstr_ae_objects>)
  target_link_libraries(fstr_ae PRIVATE fstr_core)
  set_target_properties(fstr_ae PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin PREFIX "" OUTPUT_NAME FSTR_LUT_Development)
  add_custom_command(TARGET fstr_ae POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:fstr_ae>/Contents/Resources"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_BINARY_DIR}/generated/Info.plist" "$<TARGET_BUNDLE_DIR:fstr_ae>/Contents/Info.plist"
    COMMAND /usr/bin/xcrun Rez -useDF -d __arm64__ -I "${FSTR_AE_EXAMPLES}/Headers" -I "${FSTR_AE_EXAMPLES}/Resources" "${CMAKE_CURRENT_SOURCE_DIR}/src/ae/FSTR_LUT_PiPL.r" -o "$<TARGET_BUNDLE_DIR:fstr_ae>/Contents/Resources/FSTR_LUT.rsrc"
    VERBATIM)
endif()
if(FSTR_SANITIZE AND NOT MSVC)
  target_compile_options(fstr_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(fstr_core PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(fstr_core_tests tests/core_tests.cpp)
target_link_libraries(fstr_core_tests PRIVATE fstr_core)
add_test(NAME core_contracts COMMAND fstr_core_tests)
add_executable(fstr_color_tests tests/color_tests.cpp)
target_link_libraries(fstr_color_tests PRIVATE fstr_core)
add_test(NAME color_contracts COMMAND fstr_color_tests)
if(FSTR_AE_SDK_ROOT)
  add_executable(fstr_ae_adapter_tests tests/ae_adapter_tests.cpp)
  add_dependencies(fstr_ae_adapter_tests fstr_ae)
  target_link_libraries(fstr_ae_adapter_tests PRIVATE fstr_core)
  target_include_directories(fstr_ae_adapter_tests PRIVATE src/ae "${FSTR_AE_EXAMPLES}/Headers" "${FSTR_AE_EXAMPLES}/Headers/SP" "${FSTR_AE_EXAMPLES}/Util")
  add_test(NAME owned_native_callbacks COMMAND fstr_ae_adapter_tests $<TARGET_FILE:fstr_ae>)
endif()
set(FSTR_ORT_ROOT "" CACHE PATH "External ONNX Runtime 1.23.2 root")
if(FSTR_ORT_ROOT)
  find_library(FSTR_ORT_LIBRARY NAMES onnxruntime PATHS "${FSTR_ORT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(fstr_ort src/inference/ort_backend.cpp)
  target_include_directories(fstr_ort PRIVATE "${FSTR_ORT_ROOT}/include" third_party PUBLIC include)
  target_link_libraries(fstr_ort PUBLIC fstr_core "${FSTR_ORT_LIBRARY}")
  add_executable(fstr_benchmark tools/benchmark/main.cpp)
  target_include_directories(fstr_benchmark PRIVATE third_party)
  target_link_libraries(fstr_benchmark PRIVATE fstr_ort)
endif()
