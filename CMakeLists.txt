# Versioned built-ins are usable/testable without the AE SDK. All bytes are
# checked before embedding; filenames, titles and stable menu values are owned.
set(FSTR_LOOK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/assets/looks")
file(READ "${FSTR_LOOK_ROOT}/cinema-library-v1.json" FSTR_LOOK_CATALOG)
file(SHA256 "${FSTR_LOOK_ROOT}/cinema-library-v1.json" FSTR_LOOK_BINDING)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${FSTR_LOOK_ROOT}/cinema-library-v1.json")
string(JSON FSTR_BUILTIN_COUNT LENGTH "${FSTR_LOOK_CATALOG}" entries)
math(EXPR FSTR_LAST_LOOK "${FSTR_BUILTIN_COUNT}-1")
set(FSTR_BUILTIN_ENTRIES "")
set(FSTR_BUILTIN_MENU "")
foreach(FSTR_LOOK_INDEX RANGE ${FSTR_LAST_LOOK})
  string(JSON FSTR_LOOK_VALUE GET "${FSTR_LOOK_CATALOG}" entries ${FSTR_LOOK_INDEX} menu_value)
  math(EXPR FSTR_EXPECTED_VALUE "${FSTR_LOOK_INDEX}+1")
  if(NOT FSTR_LOOK_VALUE EQUAL FSTR_EXPECTED_VALUE)
    message(FATAL_ERROR "Builtin menu values must be stable and contiguous")
  endif()
  string(JSON FSTR_LOOK_KEY GET "${FSTR_LOOK_CATALOG}" entries ${FSTR_LOOK_INDEX} key)
  string(JSON FSTR_LOOK_TITLE GET "${FSTR_LOOK_CATALOG}" entries ${FSTR_LOOK_INDEX} title)
  string(JSON FSTR_CUBE_TYPE TYPE "${FSTR_LOOK_CATALOG}" entries ${FSTR_LOOK_INDEX} cube)
  set(FSTR_CUBE_LITERAL "nullptr")
  if(NOT FSTR_CUBE_TYPE STREQUAL "NULL")
    string(JSON FSTR_CUBE_FILENAME GET "${FSTR_LOOK_CATALOG}" entries ${FSTR_LOOK_INDEX} cube)
    if(NOT FSTR_CUBE_FILENAME MATCHES "^[a-z0-9-]+\\.cube$")
      message(FATAL_ERROR "Invalid owned builtin filename")
    endif()
    string(JSON FSTR_EXPECTED_SHA GET "${FSTR_LOOK_CATALOG}" entries ${FSTR_LOOK_INDEX} sha256)
    file(SHA256 "${FSTR_LOOK_ROOT}/${FSTR_CUBE_FILENAME}" FSTR_CUBE_SHA)
    if(NOT FSTR_CUBE_SHA STREQUAL FSTR_EXPECTED_SHA)
      message(FATAL_ERROR "Frozen ${FSTR_CUBE_FILENAME} changed; create a new version")
    endif()
    file(READ "${FSTR_LOOK_ROOT}/${FSTR_CUBE_FILENAME}" FSTR_CUBE_BYTES)
    string(APPEND FSTR_LOOK_BINDING ";${FSTR_CUBE_FILENAME}:${FSTR_CUBE_SHA}")
    set(FSTR_CUBE_LITERAL "R\"FSTR_CUBE(${FSTR_CUBE_BYTES})FSTR_CUBE\"")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${FSTR_LOOK_ROOT}/${FSTR_CUBE_FILENAME}")
  endif()
  string(APPEND FSTR_BUILTIN_ENTRIES "{\"${FSTR_LOOK_KEY}\",\"${FSTR_LOOK_TITLE}\",${FSTR_CUBE_LITERAL}},\n")
  if(FSTR_LOOK_INDEX GREATER 0)
    string(APPEND FSTR_BUILTIN_MENU "|")
  endif()
  string(APPEND FSTR_BUILTIN_MENU "${FSTR_LOOK_TITLE}")
endforeach()
